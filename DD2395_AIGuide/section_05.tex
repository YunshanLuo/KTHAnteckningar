\section{Security Protocols, Key Establishment, Certificates, and TLS}

\subsection{Protocol Threat Models, Cryptographic Primitives, and Abadi--Needham Principles}
Security protocols compose cryptographic primitives over an untrusted network to achieve \term{confidentiality}, \term{authentication}, \term{integrity}, \term{freshness}, \term{explicitness (context binding)}, \term{non-repudiation}, and \term{forward secrecy}.
\begin{compactitem}
  \item \term{Dolev--Yao (Active Network) Adversary:} Controls the communication medium completely (can eavesdrop, intercept, drop, delay, replay, reorder, redirect, modify cleartext fields, and synthesize messages from known values and public keys), but \textbf{cannot break cryptographic primitives} (cannot invert $H$, decrypt without the decryption key, or forge signatures/MACs without the signing/MAC key).
  \item \term{Malicious Principal (Insider Adversary):} A legitimate protocol participant ($A$ or $B$) who holds valid long-term keys and deviates from the protocol (e.g., obtaining a server signature on a benign file $F_1$ and attaching that signature to a malicious file $F_2$).
  \item \term{Signature \& MAC Scope Invariant:} For $\mathrm{Sign}(sk_A, m) = \big(m,\; \mathrm{Enc}_{sk_A}(H(m))\big)$ verified via $\mathrm{Dec}_{pk_A}(\sigma) \stackrel{?}{=} H(m)$, integrity and origin authenticity cover \textbf{strictly the bitstring $m$ inside $H(\cdot)$}. Any field transmitted outside $m$ (e.g., $\mathrm{Sign}(sk_A, m) \concat \text{Time}$ or $\mathrm{Sign}(sk_A, \text{OK}) \concat \text{File}$) can be freely replaced by an attacker without invalidating verification.
  \item \term{Abadi--Needham Explicitness Principle:} Every signed or MACed message must explicitly state Inside the cryptographic boundary: (1) sender identity $A$, (2) intended recipient identity $B$, (3) protocol identifier and version, (4) message step number, (5) all security-relevant parameters (amount, $H(\text{File})$, key/counter), and (6) a freshness binding (nonce, timestamp, or monotonic counter).
\end{compactitem}

\subsection{Diffie--Hellman Key Exchange, PKI Trust Models, Denning--Sacco, and TLS}
\begin{compactitem}
  \item \term{Diffie--Hellman (DH) Protocol \& Discrete Logarithm Assumption:} Public parameters: large prime $q$ (e.g., 2048-bit) and primitive root (generator) $\alpha \bmod q$.
  \begin{align*}
    \text{Private keys:} \quad & X_A, X_B \xleftarrow{\$} \{1, \dots, q-2\}, &
    \text{Public keys:} \quad & Y_A = \alpha^{X_A} \bmod q, \quad Y_B = \alpha^{X_B} \bmod q, \\
    \text{Exchange:} \quad & A \to B : Y_A, \quad B \to A : Y_B, &
    \text{Shared Secret:} \quad & K_{AB} = (Y_B)^{X_A} \bmod q = (Y_A)^{X_B} \bmod q = \alpha^{X_A X_B} \bmod q.
  \end{align*}
  \term{Computational Diffie--Hellman (CDH) / DDH Hardness:} Given $(\alpha, q, \alpha^{X_A} \bmod q, \alpha^{X_B} \bmod q)$, computing or distinguishing $\alpha^{X_A X_B} \bmod q$ from random is infeasible.
  \item \term{Man-in-the-Middle (MitM) on Unauthenticated DH:} Because $Y_A, Y_B$ carry no authentication, active attacker Carol ($C$) picks private exponents $X_{C_a}, X_{C_b}$ with public values $Y_{C_a} = \alpha^{X_{C_a}} \bmod q$ and $Y_{C_b} = \alpha^{X_{C_b}} \bmod q$, executing:
  \[
    A \longrightarrow C : Y_A, \qquad C \longrightarrow B : Y_{C_b}, \qquad B \longrightarrow C : Y_B, \qquad C \longrightarrow A : Y_{C_a}.
  \]
  \emph{Outcome:} Alice computes $K_{AC} = (Y_{C_a})^{X_A} \bmod q$ (shared with Carol); Bob computes $K_{BC} = (Y_{C_b})^{X_B} \bmod q$ (shared with Carol). Carol decrypts, inspects/modifies, and re-encrypts all traffic between $A$ and $B$. \textbf{Repair:} Authenticate DH public values using long-term signing keys ($\mathrm{Sign}(sk_A, (A, B, Y_A, N_A))$) backed by CA certificates.
  \item \term{Forward Secrecy (PFS):} Compromise of a principal's long-term private key $sk_A$ in the future must \textbf{not} expose session keys of past recorded sessions. Ephemeral Diffie--Hellman (DHE/ECDHE, where $X_A, X_B$ are erased after the session) achieves Forward Secrecy; static RSA key transport ($\mathrm{Enc}_{pk_S}(PS)$) does \emph{not}.
  \item \term{Denning--Sacco Key Distribution Flaws:} Classic CA-assisted distribution $A \to B : \mathrm{Cert}_A, \mathrm{Cert}_B, \mathrm{Enc}_{sk_A}\big(\mathrm{Enc}_{pk_B}(K_{AB}, T_A)\big)$ fails because the inner ciphertext omits $A$'s identity, letting Bob reuse $\mathrm{Enc}_{sk_A}(\dots)$ or impersonate Alice to third parties if layers are re-wrapped; furthermore, leaking one predictable session key $K$ compromises related keys.
  \item \term{Public-Key Trust Models:}
  \begin{compactitem}
    \item \emph{Certificate Authority (PKI / Hierarchical Chain of Trust):} A CA signs certificate $\mathrm{Cert}_A = \mathrm{Sign}(sk_{\text{CA}}, (A, pk_A, \text{validity}))$. Browsers/OSs pre-install self-signed \term{Root CA} public keys (part of the TCB); intermediate CAs form a chain $\text{Root} \to \text{CA}_1 \to \text{Server}$. Compromise of a trusted CA allows issuing rogue certificates for any domain.
    \item \emph{Web of Trust (Decentralized, e.g., PGP):} No central root authority; users maintain keyrings and sign public keys of peers they personally verify, relying on transitive trust paths.
  \end{compactitem}
  \item \term{Transport Layer Security (TLS) Handshake \& Verification Models:}
  \begin{enumerate}[label=\arabic*., leftmargin=1.5em]
    \item \emph{ClientHello:} Client $B$ sends 256-bit random nonce $R_B$ and supported cipher suites.
    \item \emph{ServerHello + Certificate:} Server $S$ sends 256-bit random nonce $R_S$, selected cipher suite, and $\mathrm{Cert}_S$.
    \item \emph{Key Exchange \& Master Secret Derivation:} Client verifies $\mathrm{Cert}_S$, establishes premaster secret $PS$ (via $\mathrm{Enc}_{pk_S}(PS)$ in RSA mode, or signed ephemeral DH in DHE/TLS 1.3), and both parties derive directional symmetric keys $(C_B, C_S)$ for encryption and $(I_B, I_S)$ for MACs from $(R_B, R_S, PS)$.
    \item \emph{Finished (Transcript Authentication):} Both parties exchange MACs computed over the entire handshake transcript, detecting any cipher-suite downgrade or MitM tampering; fresh $(R_B, R_S)$ prevent replay.
  \end{enumerate}
  \term{Real-World TLS Vulnerabilities \& Verification:} \textbf{Downgrade attacks (FREAK/Logjam)} force legacy 512-bit export RSA/DH ciphers over unauthenticated handshake negotiation; \textbf{Apple \code{goto fail}} duplicated a \code{goto} line, bypassing signature verification on ServerKeyExchange; \textbf{Heartbleed} over-read server heap memory via unchecked TLS heartbeat length. \textbf{Symbolic verification (Tamarin)} models primitives as ideal algebraic black boxes under Dolev--Yao (fast, finds structural logic flaws, misses side channels/bit-level flaws); \textbf{Computational verification (EasyCrypt)} proves probabilistic reductions to mathematical hardness assumptions.
\end{compactitem}
