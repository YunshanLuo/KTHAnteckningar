\subsection{How to build a tree}
\begin{enumerate}
    \item Given a set of labeled data
    \item Choose the question with the maximum \textbf{information gain}
    \item \textbf{Terminate} when all data in a node have the same label or when there are no more questions to ask
\end{enumerate}

\subsection{Entropy}
Information gain is measured by shannon information content:
\[
log_2(\frac{1}{p_i}), \hspace{10pt} p_i = \text{Probability of event }i
\]

\noindent Entropy measures how predictable a dataset is and is defined as:
\[
Entropy = \sum -p_i log_2(p_i)
\]

\subsection{Entropy applied to decision trees}
Ask about the attribute that maximizes the reduction of entropy

\subsection{Information gain}
Ask about attribute $A$ for a dataset $S$ that has entropy 
$Ent(S)$ and get subsets $S_v$ according to the value of $A$
\vspace{10pt}

\noindent Information gain is defined as:
\[
Gain = \underbrace{Ent(S)}_{\text{Initial entropy}} - \sum_{v \in values(A)} \frac{|S_v|}{|S|} \hspace{-12pt}{\underbrace{Ent(S_v)}_{\text{Entropy after split}}}
\]

\subsection{Gini impurity} 
Gini impurity measures the predictability of a dataset.

\[
Gini = \sum_{i=1}p_i(1-p_i) = 1 - \sum_{i=1}p_i^2
\]
Represents the expected error rate of a random node from the tree. The lower the Gini impurity, the better the split.

\section{Challenges in ML}
How to determine the right model $f$ from data given:
\vspace{5pt}
\[
\mathcal{D} = \{(x_1, y_1), (x_2, y_2), \ldots, (x_n, y_n)\}
\]
\begin{itemize}
    \item $\mathbf{x_i} \in \text{input space}$
    \item $\mathbf{y_i} \in \text{output space}$
\end{itemize}

\textbf{Compute} the misclassification rate on $\mathcal{D}$
\[
err(f,D) = \frac{1}{N} \sum_{i=1}^{N} Ind(f(x_i) \neq y_i)
\]
\textbf{Note:} $Ind(x)$ is the indicator function, which returns 1 if the condition is true and 0 otherwise.

\subsection{Bias-variance trade off}
\begin{itemize}
    \item \textbf{Bias error}: The difference between the average prediction and the correct value
    \item \textbf{Variance error}: The variability of a model predicted datapoints
\end{itemize}

\subsection{Bias-variance decomposition}
\begin{itemize}
    \item $f(\mathbf{x})$ -- true function
    \item $\hat{f}_{\mathcal{D}}$ -- prediction function 
\end{itemize}

Expected output value:
\[
E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) \right] \text{,\hspace{10pt} The average of models output over different sample sets}
\]

Mean squared error for estimating $f(\mathbf{x})$:
\[
E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) - f(\mathbf{x}) \right]^2 = \underbrace{E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) - E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) \right]^2 \right]}_{\text{Variance error}} + \underbrace{\left( E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) \right] - f(\mathbf{x}) \right)^2}_{\text{Bias error}}
\]
\[
= \text{Variance} + \text{Bias}^2
\]
To simplify:
\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{bias_variance_decomposition.png}
\end{figure}

\subsection{Bias definition}
Measures how much the average output differs from gold standard output
\[
E[\hat{f}_{\mathcal{D}}(x)] - f(\mathbf{x})
\]
\subsection{Variance}
Measures excepted divergence of the output from the average output
\[
E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) - E_{\mathcal{D}} \left[ \hat{f}_{\mathcal{D}}(\mathbf{x}) \right]^2 \right]
\]

\section{Lecture 4 -- Regression}
The goal of regression is to predict target values for any input.

\subsection{Linear Regression}
The prediciton function is defined as:
\[
\hat{f}(\mathbf{x}) = \sum_{i=1}^{d} w_i x_i = \mathbf{w}^T \mathbf{x}
\]
We measure how well $\hat{f}(x)$ approximates $f(x)$ by minimizing the mean squared error (MSE):
\[
E_{\text{in}}(\hat{f}) = \frac{1}{N} \sum_{i=1}^{N} \left( \hat{f}(\mathbf{x}_i) - y_i \right)^2
\]
$E_{in}(w)$ can be expressed as:
\[
E_{in}(w) = \frac{1}{N} \sum^{N}_{i=1} \left( \mathbf{w}^T \mathbf{x}_i - y_i \right)^2 = \frac{1}{N} ||\mathbf{Xw} - \mathbf{y}||^2
\] 

\subsection{Residual sum of squares}
The sum of squares errors is a convex function of $w$
\[
E_{in}(w) = ||Xw - Y||^2
\]
The gradient with respect to the weights is:
\[
\frac{\partial}{\partial w}E_{in}(w) = 2X^{T}(Xw - Y)
\]
The weight vector that sets the gradient to zero is:
\[
w = (X^{T}X)^{-1}X^{T}Y
\]
\[
X^T Xw = X^T Y
\]
That is the vector that mimimizes the errors 

\subsection{RANSAC: random sampling consensus}
Repeat $M$ times:
\begin{enumerate}
    \item Sample two points to estimate a the line  
    \item Calculate the number of inliers or posterior likelihood for relation  
    \item Choose relation to maximize the number of inliers
\end{enumerate}
Maximize the number of inliers within a threshold distance
\begin{figure}[H]
    \centering
    \includegraphics[width=0.75\textwidth]{ransac.png}
\end{figure}

RANSAC algorithm:
\begin{enumerate}
    \item Randomly select sample of $s$ points to initiate subset 
    \item Find the data points $S_i$ that are within threshold distance $t$
    \item If the subset of $S_i$ is greater than threshold $T$, re-estimate the model using all points in $S_i$ and terminate
    \item If the subset of $S_i$ is less than threshold $T$, select a new subset and repeat
    \item After $N$ iterations the largest $S_i$ is selected and the model is re-estimated using all the points in the subset $S_i$
\end{enumerate}

Keep in mind that RANSAC is sensitive to the threshold distance
\begin{itemize}
    \item Too large $\implies$ multiple models will be accepted
    \item Too small $\implies$ unstable fit
\end{itemize}

\subsection{MLESAC}
MLESAC (Maximum Likelihood Estimation for Sample Consensus) is an extension of RANSAC that uses a probabilistic approach to estimate the model parameters. It assumes that the data is corrupted by noise and that the noise follows a specific distribution, typically Gaussian. The goal is to find the model that maximizes the likelihood of the observed data.

The MLESAC algorithm works as follows:
\begin{enumerate}
    \item Randomly select a sample of $s$ points to initiate the subset.
    \item Estimate the model parameters using the selected points.
    \item Calculate the likelihood of each point given the estimated model.
    \item Select the points with the highest likelihood as inliers.
    \item If the number of inliers exceeds a threshold, re-estimate the model using all inliers and terminate.
    \item If not, select a new subset and repeat.
    \item After $N$ iterations, select the model with the highest overall likelihood.
\end{enumerate}

MLESAC is more robust than RANSAC in handling noisy data, but it requires more computational resources due to the additional steps involved in calculating likelihoods.

\subsection{Training linear seperator}
Training means finding the best weights $w_i$ with two good algorithms:
\begin{itemize}
    \item \textbf{Perceptron learning algorithm:} Update weights based on misclassified examples
    \item \textbf{Delta rule:} Update weights based on the error between predicted and actual output
\end{itemize}

\textbf{Perceptron learning}
Perceptron learning has binary output
\begin{itemize}
    \item \textbf{Incremental learning:} Update weights after each example
    \item Weights only change when the example is misclassified
    \item \textbf{Update rule:} $w_i \leftarrow + \nabla (t - o)x_i$
    \item \textbf{Convergence:} If the data is linearly separable the algorithm always converges 
\end{itemize}

\textbf{Delta rule (LMS-rule)}
Delta rule has continuous output
\begin{itemize}
    \item \textbf{Incremental learning:} Update weights after each example
    \item Weights always change after each example
    \item $w_i \leftarrow + \nabla (t - o)x_i$
    \item Will find the optimal solution even if the data is not linearly separable
    \item \textbf{Convergance} only in the mean
\end{itemize}

\subsection{Linear seperation}
Hyperplane with margins
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{hm.png}
\end{figure}
\begin{itemize}
    \item Wide margins make hyperplane more generalizable
    \item Narrow margins could be overfitting the data
\end{itemize}

Mathematical formulation
\[
\vec{w}^T \vec{x} + b = 0
\]
\[
\begin{cases}
    \vec{w}^T \vec{x} \geq 1, \hspace{10pt} t = 1 \hspace{10pt} \text{(positive target)}\\
    \vec{w}^T \vec{x} \leq -1, \hspace{10pt} t = -1 \hspace{10pt} \text{(negative target)}
\end{cases}
\]

\subsection{Margin width determination}
given two ppoint $\vec{p}$ and $\vec{q}$ on the margin, the distance between them along $\vec{w}$ is:
\[
2d = \frac{\vec{w}^T}{||w^T||}(\vec{p} - \vec{q}) = \frac{2}{||\vec{w}||}
\]
maximal margin depends on the minimal length of $\vec{w}$
\vspace{10pt}


\textbf{Finding} best hyperplane
\begin{itemize}
    \item Minimize $||\vec{w}||$ 
    \item Contraints: $t_i \vec{w}^T \vec{x_i} \geq 1,$ \hspace{5pt}$\forall i$
\end{itemize}

        \item Works well with few training examples

    \end{itemize}

\subsection{Kernels}
Ultilizing high dimensional space advantage without actually representing anything high dimensional example:


\[
\text{Points in 2D:}
\hspace{15pt}
\vec{x} = \begin{bmatrix} x_1 \\ x_2 \end{bmatrix}
\]

\[
\text{Transformation to 5D}
\hspace{15pt}
\Phi(\vec{x}) = \begin{bmatrix} x_1^3 \\ x_2^2 \\ \sqrt{2}x_1x_2 \\ \sqrt{2}x_1 \\ \sqrt{2}x_2 \\ 1 \end{bmatrix}
\]

Generally expressed as:
\[
\Phi(\vec{x})^T \Phi(\vec{y}) = \kappa(\vec{x}, \vec{y})
\]
Common kernels:
\begin{itemize}
    \item Linear kernel: $\kappa(\vec{x}, \vec{y}) = \vec{x}^T \vec{y}$
    \item Polynomial kernel: $\kappa(\vec{x}, \vec{y}) = (\vec{x}^T \vec{y} + 1)^p$
    \item Radial basis function (RBF) kernel: $\kappa(\vec{x}, \vec{y}) = e^{-\frac{1}{2 \rho^2} ||\vec{x} - \vec{y}||^2}$
\end{itemize}

\subsection{Structural risk minimization}
Lagranges multiplier method
\[
L = \frac{1}{2} \vec{w}^T \vec{w} - \sum_{i=1}^n \alpha_i (t_i \vec{w}^T \vec{x_i} - 1)
\]
\begin{itemize}
    \item Minimize $L$ with respect to $\vec{w}$
    \item Maximize $L$ with respect to $\alpha_i$
\end{itemize}

\[
\frac{\partial L}{\partial \vec{w}} = 0 
\]
\[
\implies \vec{w} = \sum_i \alpha_i t_i \Phi(\vec{x_i})
\]
\[
\implies L = \sum_i \alpha_i - \frac{1}{2} \sum_{i,j} \alpha_i \alpha_j t_i t_j \Phi(\vec{x}_i)^T \Phi(\vec{x}_j)
\]
Therefore maximize 
\[\implies L = \sum_i \alpha_i - \frac{1}{2} \sum_{i,j} \alpha_i \alpha_j t_i t_j \Phi(\vec{x}_i)^T \Phi(\vec{x}_j)
\]
Under constraints:
\[
\alpha_i \geq 0, \hspace{10pt} \forall i
\]

\subsection{Support vector machine step by step}
\begin{enumerate}
    \item Choose a kernel function  
    \item Compute $\alpha_i$ by solving the dual optimization problem
    \item $\vec{x_i}$ corresponding to $\alpha_i \neq 0$ are the support vectors
    \item Classify new datapoints according to $\sum_i \alpha_i t_i K(\vec{x}, \vec{x}_i)$
\end{enumerate}

\subsection{Slack}
In SVM, slack variables are introduced to allow some data points to be on the wrong side of the margin or even the hyperplane. This is useful when the data is not perfectly linearly separable.
\vspace{10pt}

With that we can reformulate the optimization problem to this:
\[
\sum_i \alpha_i - \frac{1}{2} \sum_{i,j} \alpha_i \alpha_j t_i t_j \Phi(\vec{x}_i)^T \Phi(\vec{x}_j)
\]
With the constraints:
\[
0 \leq \alpha_i \leq C, \hspace{10pt} \forall i
\]

\section{Artificial Neural Networks}
Artificial neural networks (ANNs) are a class of machine learning models inspired by the structure and function of the human brain. They consist of interconnected layers of nodes (neurons) that process input data and learn to make predictions or classifications.
\subsection{Single neuron illustration}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.75\textwidth]{single_neuron.png}
\end{figure}

\textbf{Net input} of a neuron is given by:
\[
net = \sum_{i=1}^{n} p_i w_i = p_1 w_1 + p_2 w_2 + \ldots + p_n w_n
\]
\textbf{Ouput} of the neuron is given by:
\[
output = f(net - \theta)
\]
\begin{itemize}
    \item $f$ is the activation function
    \item $\theta$ is the threshold
\end{itemize}
The XOR function cannot be represented by a single neuron, but can be represented by a multi-layer neural network.

\subsection{Common activation functions}
\begin{enumerate}
    \item Linear function: $f(x) = x$
    \item Step function: $f(x) = 1$ if $x \ge 0$, else $f(x) = 0$
    \item Ramp function: $f(x) = 0$ if $x < 0$, else $f(x) = x$
    \item Sigmoid function: $f(x) = \frac{1}{1 + e^{-x}}$
    \item Hyperbolic tangent function: $f(x) = \frac{e^x - e^{-x}}{e^x + e^{-x}}$
    \item Gaussian function: $f(x) = e^{-x^2}$
\end{enumerate}

\subsection{Training Neural Networks}
\textbf{Supervised learning} uses labeled training data. The training uses error produced by a neuron to adjust its weights. 
\newline
\vspace{10pt}

\textbf{The Error metric:}
\begin{itemize}
    \item $t_p(t)$ is the target output
    \item $out_p(t)$ is the actual output
\end{itemize}
\[
E(t) = \sum_{p=1}^{P_{\tau}} (t_p(t) - out_p(t))^2
\]

\textbf{Weight update for a neuron:}
\begin{itemize}
    \item $\eta$ is the learning rate   
    \item $\Delta w_i(t)$ is the weight change for weight $w_i$
    \item $\Delta w_i(t)$ is proportional to the negative gradient of the error with respect to the weight $w_i$
\end{itemize}
\[
w_i(t) = w_i(t - 1) + \Delta w_i (t)
\]
\[
\Delta w_i(t) = \eta (\frac{- \partial E(t)}{\partial w_i(t)})
\]

\subsection{Types of neural networks}

\subsection{How to train neural networks}
We use \textbf{back propagation} to determine the error and update the weights with \textbf{gradient descent}
\[
\Delta w_{ik}(t) = f'(net(t)) \cdot \sum_{j \in P} \delta (t) \cdot w_{kj}(t) \cdot f(net_{k}(t))
\]
\[
\delta_i (t) = f'(net_i (t)) \cdot \sum_{j \in P} \delta (t) \cdot w_{ji}(t)
\]
Step by step:
\begin{enumerate}
    \item \textbf{Weight initialization:} Initialize all the weights to small random numbers
    \item \textbf{Input/Output normalization:} Normalize the input and output data to a range of [0, 1] or [-1, 1]
    \item \textbf{Learning rate:} Use a small weight update $\eta$
    \item Avoid local minima in weight space by doing multiple runs
    \item \textbf{Network size:} fan-in and fan-out of the network should be chosen based on the complexity of the problem
    \item \textbf{Vanishing gradient:} Multiple layers significantly reduces the available gradient
    \item \textbf{Overfitting/overtraining:} Overfitting to training data can cause poor generalization
\end{enumerate}
Good pratices
\begin{itemize}
    \item Split given data into two sets:
    \begin{itemize}
        \item 70\% training set
        \item 30\% validation set
    \end{itemize}
    \item Train on training data only 
    \item Stop training when the error on the validation set starts to increase
\end{itemize}

\subsection{Kolmegorov's Axiom}
\begin{enumerate}
    \item $P(E) \geq 0$ for any event $E in \Omega$
    \item $P(\Omega) = 1$
    \item For any sequence of mutually exclusive events $E_1, E_2, \ldots$, we have $P(\cup_{i=1}^{\infty} E_i) = \sum_{i=1}^{\infty} P(E_i)$
\end{enumerate}

\begin{enumerate}
    \item $P(A) \leq P(B) if A \subseteq B$
    \item $P(A \cup B) = P(A) + P(B) - P(A \cap B)$
    \item $P(\neg A) = 1 - P(A)$
\end{enumerate}

\section{Learning as inference}
Simple terminology:
\begin{itemize}
    \item Data: $\mathcal{D} = \{(x_1, y_1), (x_2, y_2), \ldots, (x_n, y_n)\}$
    \item Prior: $p(y, \mathcal{D})$
    \item Evidence: $p(x.\mathcal{D})$
    \item Likelihood: $p(x|y, \mathcal{D})$
    \item Posterior: $p(y|x, \mathcal{D})$
\end{itemize}

Discriminative model:
\begin{itemize}
    \item Models $p(y|x. \mathcal{D})$ directly 
    \item Examples: Logistic regression, SVM, neural networks
\end{itemize}

Generative model:
\begin{itemize}
    \item Models $p(x|y, \mathcal{D})$ and $p(y|\mathcal{D})$ separately
    \item Examples: Naive Bayes, Gaussian mixture models, hidden Markov models
\end{itemize}

\subsection{Parametric inference}
\[
p(y|x) = p(y|x, \theta)
\]
\begin{enumerate}
    \item $\theta$ is a learned parameter
    \item $\theta$ is learned from $\mathcal{D}$
\end{enumerate}

\subsection{Point estimates vs Bayesian Inferene}
Point estimates:
\begin{itemize}
    \item Estimate $\theta$ using $\mathcal{D}$
    \item Compute $p(y|x, \hat{\theta})$ to make inference 
    \item Learning means to estimate $\hat{\theta}$ from $\mathcal{D}$
\end{itemize}

Distributional bayesian inference:
\begin{itemize}
    \item Estimate $p(\theta|mathcal{D})$ 
    \item Compute $p(y|x, \mathcal{D}) = \int p(y|x, \theta)p(\theta|\mathcal{D}) d\theta$ to make inference
    \item We learn a distribution over $\theta$ from $\mathcal{D}$ instead of a point estimate
\end{itemize}

\subsection{Parameter learning methods}
\begin{itemize}
    \item Maximum likelihood estimation (MLE): Find the parameter $\theta$ that maximizes the likelihood of the observed data $\mathcal{D}$
    \item Maximum a posteriori estimation (MAP): Find the parameter $\theta$ that maximizes the posterior distribution $p(\theta|\mathcal{D})$
    \item Bayesian inference: Compute the posterior distribution $p(\theta|\mathcal{D})$ and use it to make predictions
\end{itemize}

\subsection{Maximum likelihood}
Find the parameter $\theta$ that maximizes the likelihood of the observed data $\mathcal{D}$:
\[
\hat{\theta}_{MLE} = \arg \max_{\theta} p(\mathcal{D}|\theta)
\]
\subsection{Probabilistic linear regression}
Assume a linear model with gaussian noise:
\[
y = \mathbf{w}^T x + \epsilon, \hspace{10pt} \epsilon \sim \mathcal{N}(0, \sigma^2)
\]
Therefore, 
\[
Y|X = x \sim \mathcal{N}(\mathbf{w}^T x, \sigma^2)
\]
The goal is to find $\mathbf{\tilde{w}}$ that maximizes:
\[
P(\mathcal{D} | \mathbf{w}, \sigma^2)
\]
We use log maximum likelihood method to find the optimal $\mathbf{w}$:
\[
\log p(y|x,w,\sigma^2) = \sum_{i}^{} \log(p(y_i), x_i, w, \sigma^2)
\]
\[
= \sum_{i}^{} \log \left( \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left( -\frac{(y_i - \mathbf{w}^T x_i)^2}{2\sigma^2} \right) \right) 
\]
\[
\implies \arg\min_{w} \sum_{i}^{} (y_i - \mathbf{w}^T x_i)^2
\]

\subsection{Bayes classifer: MAP and ML decision rules}
Given features $\mathbf{x}$ and $\mathscr{y} = $ \{$y_1, y_2, \ldots, y_k$ \} we want to find the class $y_i$ that maximizes the posterior probability $p(y_i|\mathbf{x})$. Using Bayes theorem:

\[
\tilde{y}_{MAP} = \arg\max_{y_k} p(y_k|\mathbf{x}) = \arg\max_{y_k} \frac{p(\mathbf{x}|y_k)p(y_k)}{p(\mathbf{x})} = \arg\max_{y_k} p(\mathbf{x}|y_k)p(y_k)
\]

Because $p(\mathbf{x})$ is constant for all classes

\subsection{ML estimation of Gaussian mean}
\[
p(x|\theta) = \mathcal{N}(x|\mu, \sigma^2) = \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left( -\frac{(x - \mu)^2}{2\sigma^2} \right)
\]
\[
\log(p(x|\theta)) = \sum_{n=1}^{N} \log(\mathcal{N}(x_n|\mu, \sigma^2)) = -\frac{N}{2} \log(2\pi\sigma^2) - \frac{1}{2\sigma^2} \sum_{n=1}^{N} (x_n - \mu)^2
\]
\[
\frac{\partial}{\partial \mu} = \log(p(x|\theta)) = \frac{1}{\sigma^2} \sum_{n=1}^{N} (x_n - \mu) = 0
\]
\[
\implies \frac{1}{N} \sum_{n=1}^{N} x_n
\]

Sample mean:
\[
\bar{\mu} = \frac{1}{N} \sum_{n=1}^{N} x_n
\]

Average variance:
\[
\bar{\sigma}^2 = \frac{1}{N} \sum_{n=1}^{N} (x_n - \bar{\mu})^2

Formula for MLE:
\[
p(\mathcal{D}, \lambda) = \prod_{i=1}^{N} p(y_i|\lambda) = \prod_{i=1}^{N} \lambda^{y_i}(1 - \lambda)^{1 - y_i}
\]
\[

\]
\[
\implies log(p(\mathcal{D}|\lambda)) = n\cdot log(\lambda) + (N - n) \cdot log(1 - \lambda)

Solve the derivative for $log(p(\mathcal{D}|\lambda))$ to find the maximum $\lambda$:
\[
\frac{\partial}{\partial \lambda} log(p(\mathcal{D}|\lambda)) = \frac{n}{\lambda} - \frac{N - n}{1 - \lambda} = 0
\]
\[
\implies \frac{n - N \lambda}{\lambda (1-\lambda)} = 0
\]
\[
\implies n - N \lambda = 0 
\]
\[
\implies {\lambda_{ML}} = \frac{n}{N}

\subsection{D-dimensional zero-mean spherical gaussian prior}
Assume that $\mathbf{w} \sim \mathcal{N}(0, \sigma_p^2 I)$ is a D-dimensional zero-mean spherical Gaussian prior. The prior distribution is given by:
\[
p(\mathbf{w}|0, \sigma_p^2) = \frac{1}{(2\pi\sigma_p^2)^{D/2}} \exp\left(-\frac{||\mathbf{w}||^2}{2\sigma_p^2}\right)
\]
We optimize the log posterior $log(\mathbf{w} | y, x)$
\[
w_{MAP} = \text{arg max}_{\mathbf{w}}log(p(\mathbf{w}|y, x)) = \text{arg max}_{\mathbf{w}} \left[ log(p(\mathbf{w})) + log(p(y|x, \mathbf{w})) \right]
\]
\[
= \text{arg min}_{\mathbf{w}} \left[ \underbrace{\frac{1}{2\sigma^2_y} \sum_{n=1}^{N}(y_n - \mathbf{w}^ x_n)^2}_{\text{data fit}} + \overbrace{\frac{1}{2\sigma_p^2} ||\mathbf{w}||^2}^{\text{ridge regression term}} \right]
\]
Ridge regression is a regularization technique that adds a penalty term to the loss function to prevent overfitting.

\subsection{IID (independent and identically distributed) zero-mean laplace for each dimension}
Occurs when $p(\mathbf{w} | b_w) = \prod_{d}^{} \frac{1}{2b_w} exp(-\frac{|w_d|}{b_w})$
where $b_w$ is the scale parameter and $w_d$ is the d-th dimension of $\mathbf{w}$. $w_{MAP}$ is given by:
\[
w_{MAP} = \text{arg max}_{w} log(p(\mathbf{w}|y, x)) = \text{arg max}_{\mathbf{w}} \left[ log(p(\mathbf{w})) + log(p(y|x, \mathbf{w})) \right]
\]

\[
\text{arg min}_{w} \left[ \underbrace{\frac{1}{2\sigma^2_y} \sum_{n=1}^{N}(y_n - \mathbf{w}^T x_n)^2}_{\text{data fit}} + \overbrace{\frac{1}{b_w} \sum_{d=1}^{D} \mathbf|{w_d}|}^{\text{lasso regression term}} \right]
\]
Lasso is a penalization technique that adds a penalty term to the loss function to encourage sparsity in the model parameters.

\subsection{Bayesian parameter inference}
\begin{itemize}

    \item Compute new predicting posterior by marginalizing over $\theta$:

\subsection{Bayesian Linear Regression}
Given $\mathcal{D} = \{(\mathbf{x_1}, y_1), (\mathbf{x_2}, y_2), \ldots, (\mathbf{x_N}, y_N)\}$
\begin{itemize}
    \item $Y1,.., Y_N$ are conditionally independent given $\mathbf{w}$
    \item $Y|X \sim \mathcal{N}(\mathbf{w}^T \mathbf{x}, \sigma^2)$
    \item $\mathbf{w} \sim \mathcal{N}(\mathbf{\mu}_\mathbf{w}, \mathbf{\Sigma}_{\mathbf{w}}), \hspace{10pt} \mathbf{w} \in \mathbb{R}^\mathcal{D}$
    \item We assume values for $\sigma^2, \mathbf{\mu}_\mathbf{w}, \mathbf{\Sigma}_{\mathbf{w}}$ and assume $\theta \equiv \mathbf{w}$
    \item The goal is to estimate $P(y|x, \mathcal{D})$
\end{itemize}

\[
P(y, \mathbf{x}, \mathcal{D}) = \int_{\mathbf{w} \in \mathbb{R}^\mathcal{D}}^{} P(y, \mathbf{x}, \mathcal{D} | \mathbf{w}) P(\mathbf{w}) d\mathbf{w}
\]
\begin{itemize}
    \item $P(y, \mathbf{x}, \mathbf{w})$ is Gaussian therefore $P(y, \mathbf{x}, \mathcal{D})$ must be Gaussian as well
\end{itemize}

Step by step guide:
\begin{itemize}
    \item \textbf{Define:} 
    \begin{itemize}
        \item $\Phi^T = [\mathbf{x_1}, \mathbf{x_2}, \ldots, \mathbf{x_N}]$
        \item $\mathbf{y}^T = [y_1, y_2, \ldots, y_N]$
        \item $W|D \sim \mathcal{N}(\mathbf{\mu}_{\mathbf{w}|\mathcal{D}}, \mathbf{\Sigma}_{\mathbf{w}|\mathcal{D}})$
    \end{itemize}
\end{itemize}

\[
\implies \Sigma_{\mathbf{w}|\mathcal{D}} = \left[ \frac{1}{\Sigma_{\mathbf{w}}} + \frac{\Phi^T \Phi}{\sigma_{y}^2}\right]^{-1}
\]
\[
\implies \mu_{\mathbf{w}|\mathcal{D}}= \Sigma_{\mathbf{w}|\mathcal{D}} \left[\frac{1}{\Sigma_{\mathbf{w}}} \mathbf{\mu}_{\mathbf{w}} + \frac{\Phi^T \mathbf{y}}{\sigma_{y}^2}\right]
\]

The predictive posterior distribution is given by:
\[
p(y|x, \mathcal{D}) = \mathcal{N}(\mu_{y|x, \mathcal{D}}\mathbf{x}, \sigma^2_{y} + \mathbf{x}^T \Sigma_{\mathbf{w}|\mathcal{D}} \mathbf{x})
\]
The uncertainty over the choice of $\mathbf{w}$ and $\sigma_Y^2$ prevents overfitting and allows for better generalization to unseen data.

\subsection{Occam's Razor}
\begin{itemize}
    \item When faced with multiple explanations for the same phenomenon, the simplest explanation is preferred
    \item In machine learning, this means that simpler models are preferred over more complex models,
    \item This is because simpler models are less likely to overfit the training data and are more likely to generalize well to unseen data.
\end{itemize}

\subsection{Overfitting and Maximum Likelihood}
More flexible models can achieve higher maximum likelihood however, they are more likely to overfit the training data and perform poorly on unseen data. This is because flexible models can fit the noise in the training data rather than the underlying pattern.

\subsection{Limitations of bayesian inference}
A flexible model may fit the data extremely well, but only for a small range of parameter values. A good fit over many values implies high evidence
\[
P(y|\mathbf{x},\mathcal{D}) = \int_{\mathbb{R^\mathcal{D}}}^{} P(y|\mathbf{x},\mathbf{w})P(\mathbf{w}|\mathcal{D}) d\mathbf{w}
\]
\begin{itemize}
    \item A closed form posterior $P(\mathbf{w}|\mathcal{D})$ is not always available
    \item We can use approximation such as sampling methods like MCMC or complex solutoins
    \item A non-informative prior provides little prior guidance but bayesian inference still represents parameter uncertainty
\end{itemize}

    $\mathcal{D}=\{\boldsymbol{x}_i\in\mathbb{R}^d\}$,
    \Statex \hspace{3.2em} distance function
    $f(\mathbf{x},\boldsymbol{\mu})=\lVert\mathbf{x}-\boldsymbol{\mu}\rVert_2^2$
    \State \textbf{initialization:} sample $K$ centroids
    $\boldsymbol{\mu}_k^{(0)}\in\mathbb{R}^d$ at random; $t\gets 0$;
    \Repeat
        \State assign each point $x_i$ to closest centroid:
        \[
        \mathcal{C}_k \coloneqq \{\mathbf{x}\in\mathcal{D}:\,
        k = \arg\min_{k'} f\!\left(\mathbf{x},\,\boldsymbol{\mu}_{k'}^{(t)}\right)\}
        \]
        \State update centroids as mean of each cluster of points:
        \[
        \boldsymbol{\mu}_k^{(t+1)} = \frac{1}{|\mathcal{C}_k|}
        \sum_{\mathbf{x}\in\mathcal{C}_k}\mathbf{x}
        \]
        \State iterate $t \gets t+1$
    \Until{$\displaystyle\max_k f\!\left(\boldsymbol{\mu}_k^{(t-1)},\,
    \boldsymbol{\mu}_k^{(t)}\right) < \epsilon$;}
    \State \textbf{return} $K$ clusters and their centroids
\end{algorithmic}

K-means limitations:
\begin{itemize}
    \item All distances are equally weighted
    \item Favors spherical clusters
    \item Cluster boundraies are determined by distances to the centroids
\end{itemize}

\subsection{Fitting complex distributions to K-means}
We can fit a mixture of $K$ distributions:
\[
p(\mathbf{x}|\theta) = \sum_{k=1}^{K} \pi_k p(\mathbf{x}|\theta_k)
\]
with $\theta = \{\pi_1,...,\pi_k,\theta_1,...,\theta_k\}$
\begin{itemize}
    \item $\pi_k$ is the mixing coefficient for the k-th distribution
    \item $\theta_k$ is the parameter for the k-th distribution
    \item $p(\mathbf{x}|\theta_k)$ is the probability density function for the k-th distribution
\end{itemize}

\subsection{Expectation Maximization (EM)}
Fitting model parameters with missing latent variables
\[
p(\mathbf{x}|\theta) = \sum_{k=1}^{K} \pi_k p(\mathbf{x}|\theta_k)
\]
with $\theta = \{\pi_1,...,\pi_k,\theta_1,...,\theta_k\}$
\begin{itemize}
    \item Introduce latent variable $h_i = \left\{1,...,K\right\}$ to indicate which distribution generated $\mathbf{x}_i$
\end{itemize}
EM steps alteratively between:
\begin{itemize}
    \item \textbf{E-step:} Estimate the the probabilites of the hidden assignments
    \item \textbf{M-step:} Update $\theta$ using those probabilites
\end{itemize}

Assume that distribution is a mixture of $K$ gaussian distributions

\[
p(x) = \sum_{k=1}^{K} \pi \mathcal{N}(x|\mu_k, \Sigma_k), \hspace{10pt} \sum_{k=1}^{K} \pi_k = 1
\]

Interpret the gaussian mixture with a latent variable $h_i \in \{1, 2, \ldots, K\}$ that indicates which gaussian generated $x_i$:
\[
p(x) = \sum_{k=1}^{K} p(h_i = k)p(x|h_i = k) = \sum_{k=1}^{K} \pi_k \hspace{2pt} \mathcal{N} \hspace{1pt}(x|\mu_k, \Sigma_k)
\]


\subsection{EM for two Gaussians}
\[
h_i = \begin{cases}
    1, \hspace{10pt} \text{if sample} x_i \text{was drawn from} \mathcal{N}(\mu_1, \Sigma_1) \\
    2, \hspace{10pt} \text{if sample} x_i \text{was drawn from} \mathcal{N}(\mu_2, \Sigma_2)
\end{cases}
\]
Come up with initial guesses for:
\[
\theta^{(0)} = \{\pi_1^{(0)}, \mu_1^{(0)}, \sigma_1^{(0)}, \pi_2^{(0)}, \mu_2^{(0)}, \sigma_2^{(0)}\}
\]
EM-algorithm will iteratively update the parameters $\theta$ until convergence

\subsection{E-step}
Compute the posterior probability that $x_i$ was generated by the k-th gaussian:
\vspace{15pt}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{e-step.png}
\end{figure}

for each data point $i = 1, \ldots, N$

\quad for each mixture component $k = 1, 2$

\begin{align*}
\gamma_{ik}^{(t)} &\coloneqq p(h_i = k \mid x_i, \Theta^{(t)}) = \frac{p(h_i = k \mid \Theta^{(t)}) \, p(x_i \mid h_i = k, \Theta^{(t)})}{p(x_i \mid \Theta^{(t)})} \\
&= \frac{\pi_k^{(t)} \, \mathcal{N}\!\left(x_i \mid \mu_k^{(t)}, \sigma_k^{(t)}\right)}{\pi_1^{(t)} \, \mathcal{N}\!\left(x_i \mid \mu_1^{(t)}, (\sigma_1^{(t)})^2\right) + \pi_2^{(t)} \, \mathcal{N}\!\left(x_i \mid \mu_2^{(t)}, (\sigma_2^{(t)})^2\right)}
\end{align*}

\[
\gamma_{i1}^{(t)} + \gamma_{i2}^{t} = 1 
\]
\[
\text{mixture weights: } \pi_1 + \pi_2 = 1
\]


\subsection{M-step}
Fitting the Gaussian model for each component 
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{m-step.png}
\end{figure} 

Compute the maximum likelihood estimates of the parameters $\theta$ using the posterior probabilities $\gamma_{ik}^{(t)}$ computed in the E-step:

\vspace{10pt}

for $k = 1, 2$
\[
\mu_k^{(t+1)} = \frac{\sum_{i=1}^{N} \gamma_{ik}^{(t)} \, x_i}{\sum_{i=1}^{N} \gamma_{ik}^{(t)}}
\]

\[
\sigma_k^{(t+1)} = \sqrt{\frac{\sum_{i=1}^{N} \gamma_{ik}^{(t)} \left(x_i - \mu_k^{(t+1)}\right)^2}{\sum_{i=1}^{N} \gamma_{ik}^{(t)}}}
\]

\[
\pi_k^{(t+1)} = \frac{\sum_{i=1}^{N} \gamma_{ik}^{(t)}}{N}
\]

\subsection{EM vs K-means}
Similarities:
\begin{itemize}
    \item Each iteration does not decrease the observed data likelihood
    \item Converges to a stationary point and is sensitive to inital conditions
\end{itemize}

More flexible than K-means:
\begin{itemize}
    \item \textbf{Full covariance} gaussian components can model clusters with different ellipsodial sizes and orientations
    \item \textbf{Soft assignments} of points to clusters allows for more flexible cluster
    \item Can include prior $p(\theta)$ to obtain MAP estimates of the parameters
\end{itemize}

\subsection{Bagging (Bootstrap Aggregating)}
We train 6 detectors on the same dataset with random subsets of the data (resampled dataset)

\textbf{High bias, Low bias classifiers e.g. decision trees}:
\begin{itemize}
    \item \textbf{High variance:} Classifiers produce decision boundaries highly sensitive to training data. 
    \item \textbf{Low bias:} classifiers produce decision boudaries which on average don't differ much from gold output
\end{itemize}

\subsection{Bagging input/ouput}
\textbf{Input:} Training data
\[
\mathcal{S} = \left\{(\mathbf{x_1}, y_1), (\mathbf{x_2}, y_2), \ldots, (\mathbf{x_N}, y_N)\right\}
\]
$\mathbf{x}_j$ are feature vectors, $y_j$ are labels
\vspace{10pt}

\textbf{Iterate:} For b = 1, 2, ..., B
\begin{enumerate}
    \item Sample training examples, with replacement, $m'$ times from $\mathcal{S}$ to create $S_b(m' \leq m)$
    \item Use this bootstrap sample $\mathcal{S}$ to estimate the regression or classification function $f_b$
\end{enumerate}

\textbf{Output:} The bagging estimate for
\vspace{10pt}

\textbf{Classification:}
\[
f_{bag}(\mathbf{x}) = \arg \max_{1 \leq k \leq K} \sum_{b=1}^{B} Ind(f_b(\mathbf{x}) = k)
\]

\textbf{Regression:}
\[
f_{bag}(\mathbf{x}) = \frac{1}{B} \sum_{b=1}^{B} f_b(\mathbf{x})
\]

\textbf{Indicator function:}
\[
Ind(x) = \begin{cases}
    1, \hspace{10pt} x = 1 \\
    0, \hspace{10pt} \text{otherwise}
\end{cases}
\]

Bagging is a \textbf{reduces variance} of the classifier especially when labelled data is limited. Bagging should only be used for \textbf{high variance, low bias} classfiers. Example illustration

\subsection{Boosting}
Using a set of weak learners to create a strong classifier

\vspace{10pt}
\textbf{Example}
\begin{itemize}
    \item A set of high bias classifiers combined can produce a low bias classifier
    \item We do this by voting
\end{itemize}

Assume $\mathcal{H} = \{h_1, h_2, \ldots, h_T\}$ are weak classifiers \textbf{voting is:}
\[
sign \left(\sum_{t=1}^{T}h_t(\mathbf{x})\right)
\]

\subsection{Boosting input output}
\textbf{Input:} Training data $\mathcal{S} = \left\{(\mathbf{x_1},y_1),...,(x_m,y_m)\right\}$, $\mathbf{x_i}$ are feature vectors, $y_i$ are labels, $y_i \in \{-1, 1\}$  
\newline

\vspace{10pt}
\textbf{Output: A strong classifier/regression function:}
\[
f_T(\mathbf{x}) = sign\left(\sum_{t=1}^{T} \alpha_t h_t(\mathbf{x})\right)
\]
or
\[
f_T(\mathbf{x}) = \sum_{t=1}^{T} \alpha_t h_t(\mathbf{x})
\]
This is the weighted sum of weak classifiers where $\alpha_t$ is called confidence/reliability 
\begin{figure}[H]
    \centering
    \includegraphics[width=0.75\textwidth]{boosting.png}
\end{figure}

Summary boosting:
\begin{itemize}
    \item A set of classifiers are combined to produce a strong classifier
    \item Maintain \textbf{weight} $w_i^{(t)}$ for each training example in $\mathcal{S}$
    \item Large $w_i^{(t)}$ means that $\mathbf{x_i}$ has greater influence on choice of $h_t$ 
    \item Iteration: $w_i^{(t+1)}$ is updated based on whether $h_t$ correctly classified $\mathbf{x_i}$
\end{itemize}

\subsection{Adaboost algorithm}
Given label training data $\mathcal{S} = \left\{(\mathbf{x_1},y_1),...,(x_m,y_m)\right\}$, $\mathbf{x_i} \in \mathbb{R}^d$ are feature vectors, $y_i$ are labels, $y_i \in \{-1, 1\}$
\newline
\vspace{10pt}

A set of potentially weak classifiers $\mathcal{H} = \{h_1, h_2, \ldots, h_T\}$
\vspace{10pt}

\textbf{Initializaiton:}
Introduce a weight $w_i^1$ for each training sample $\mathbf{w} \in \mathbb{R}^m$, set $w_i^1 = \frac{1}{m}$ for $i = 1, 2, \ldots, m$
\vspace{10pt}
\newpage
\textbf{The algorithm: }
\newline
\textbf{Iterate:} For $t = 1,...,T$
\begin{enumerate}
    \item Train weak classifier $h_t \in \mathcal{H}$ using $\mathcal{S}$ with weights $w_i^{(t)},...,w_m^{(t)};$ select the one that minimizes the training error:
    \[
    \epsilon_t = \sum_{j=1}^{m} w_j^{(t)} Ind(y_j \neq h_t(\mathbf{x_j}))
    \]
    \item Compute the \textbf{reliability coefficient:}
    \[
    \alpha = log \left(\frac{1-\epsilon}{\epsilon_t}\right) 
    \]
    $\epsilon$ must be less than $0.5$ break out of loop if $\epsilon_t \approx 0.5$
    \item Update the weights by using: $w_j^{(t+1)} = w_j^{(t)} \exp(-\alpha_t y_j h_t(\mathbf{x_j}))$ 
    \item Normalize the weights such that $\sum_{j=1}^{m} w_j^{(t+1)} = 1$
\end{enumerate}

\subsection{Principal Component Analysis (PCA)}
Maximizing variance:
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{pca.png}
\end{figure}

Mean vector of $x$:
\[
E(x) = \frac{1}{\underbrace{r}_{\text{number of samples}}} \sum_{i=1}^r x_i
\]

Covariance matrix of $x$:
\[
\Sigma = E((x-E(x))(x-E(x))^T)
\]

\subsection{Maximum variance criterion}

Reduce the effectie number of variables:
\[
E((x^T u_i - E(x^T u_i))^2) \rightarrow \text{Maximize for: }(i = 1, 2, \ldots, p), \hspace{10pt}\text{Condition: } u_i^T u_j = \delta_{ij} 
\]
\[
= u_i^T \underbrace{E((x - E(x))(x - E(x))^T)}_{Covariance matrix} u_i = u_i^T \Sigma u_i
\]
\[
max[tr(U^T \Sigma U)]
\]
The transformation matrix $U$ is a $d \times p$ matrix with columns $u_1, u_2, \ldots, u_p$ and $p < d$. The columns of $U$ are the eigenvectors of the covariance matrix $\Sigma$ corresponding to the $p$ largest eigenvalues. Example of dimensionality reduction:

\begin{figure}[H]
    \centering
    \includegraphics[width=0.75\textwidth]{pca2.png}
\end{figure}

Autocorrelation matrix of $x$, recall least squares regression:
\[
Q = E(xx^T)
\]

\subsection{Minimum squared distance criterion}
Average squared error between $x$ and its approximation to be minimized by a set $\left\{u_1,...,u_p\right\}$
\[
E(||x - x'||^2) \rightarrow \text{minimize} (i = 1,...,p)
\]
Approximated $x'$ is given by:
\[
x' = \sum_{p}^{i=1} (x^Tu_i)u_i
\]
Maximize:
\[
||x'||^2 = ||x||^2 - ||\tilde{x}||^2
\]

The basis consists of $p$ eigenvectors of the autocorrelation matrix $Q$ corresponding to its $p$ largst eigenvalues

\subsection{Subspace}
Subspace $L$ is a collection of span of linearly independent vectors

\[
L(b_1,...,b_p) = \left\{z|z = \sum_{i=1}^{p} \xi b_i \right\}, \hspace{10pt} (\xi \in \mathbb{R}, b_i \in \mathbb{R}^n)
\]
Dimension of a subspace:
\[
p = dim(L) << n
\]
Usually represented of orthonormal basis where the basis vectors are pairswise orthogonal
